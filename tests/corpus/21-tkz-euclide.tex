\documentclass{standalone}
% tkz-euclide: triangle with its circumcircle (50-file family).
\usepackage{tkz-euclide}
\begin{document}
\begin{tikzpicture}
\tkzDefPoints{0/0/A, 4/0/B, 2/3/C}
\tkzDrawPolygon(A,B,C)
\tkzCircumCenter(A,B,C)\tkzGetPoint{O}
\tkzDrawCircle(O,A)
\tkzDrawPoints(A,B,C,O)
\tkzLabelPoints(A,B,C,O)
\end{tikzpicture}
\end{document}
