\documentclass{article}
\usepackage{tikz}
\usepackage{algorithm}
\usepackage{algpseudocode}
\begin{document}
\begin{algorithm}[H]
\caption{Positive value}
\begin{algorithmic}[1]
\State $x\gets 1$
\While{$x>0$}
\If{$x=1$}
\State \Return $x$
\EndIf\EndWhile
\end{algorithmic}
\end{algorithm}
\end{document}
