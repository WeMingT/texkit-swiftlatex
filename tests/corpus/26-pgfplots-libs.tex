\documentclass{standalone}
% pgfplots library composite: 2x1 groupplot + date axis + axis units.
\usepackage{pgfplots}
\usepgfplotslibrary{groupplots,dateplot,units}
\pgfplotsset{compat=1.18}
\begin{document}
\begin{tikzpicture}
\begin{groupplot}[
  group style={group size=2 by 1, horizontal sep=2.6cm},
  width=6cm, height=5cm,
]
\nextgroupplot[
  title={voltage over date},
  date coordinates in=x,
  date ZERO=2026-01-01,
  xlabel={date},
  ylabel={voltage},
]
\addplot coordinates {
  (2026-01-01, 1.0)
  (2026-02-01, 2.5)
  (2026-03-01, 4.0)
};
\nextgroupplot[
  title={distance over time},
  xlabel={time},
  x unit=s,
  ylabel={distance},
  y unit=m,
]
\addplot[domain=0:4, samples=17] {x^2};
\end{groupplot}
\end{tikzpicture}
\end{document}
